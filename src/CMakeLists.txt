cmake_minimum_required(VERSION 3.18)
project(incgraph LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(INCGRAPH_NATIVE "Compile with -march=native" ON)

find_package(OpenMP REQUIRED)

add_library(incgraph_core STATIC core/index.cpp core/util.cpp core/policies.cpp)
target_include_directories(incgraph_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(incgraph_core PUBLIC OpenMP::OpenMP_CXX)
set_target_properties(incgraph_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(incgraph_core PRIVATE -Wall -Wextra -Wno-sign-compare)
if(INCGRAPH_NATIVE)
  target_compile_options(incgraph_core PUBLIC -march=native)
endif()

set(BIN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../bin)
foreach(app ig_build ig_make_mm ig_make_coco ig_make_runbook ig_make_drift ig_runbook ig_drift ig_curve ig_crossmodal)
  add_executable(${app} apps/${app}.cpp)
  target_link_libraries(${app} PRIVATE incgraph_core)
  target_compile_options(${app} PRIVATE -Wall -Wextra -Wno-sign-compare)
  set_target_properties(${app} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${BIN_DIR})
endforeach()
add_executable(ig_test tests/ig_test.cpp)
target_link_libraries(ig_test PRIVATE incgraph_core)
set_target_properties(ig_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${BIN_DIR})
